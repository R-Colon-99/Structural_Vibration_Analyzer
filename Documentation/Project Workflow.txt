Project Workflow

conda activate Mech-Engi-env
mkdir "F:"				// Create new file location. If location already exists, can skip this step

cd "Project_Name"			// open project folder
mkdir data				// make data folder
mkdir exports				// make exported data folder
mkdir scripts				// make folder for script versions
code .					// open VS code to edit scripts
python scripts\"script_name.py"		// run the script
